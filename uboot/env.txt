# U-Boot environment for edison-image, written to u-boot-env0 and u-boot-env1 by the
# flash script (mkenvimage -r -s 0x10000). It replaces mainline U-Boot's default Edison
# environment, which dates from the vendor kernel (console=ttyMFD2, root by saved
# partition UUID, OTA/rescue logic we don't use).
#
# Boot flow: make sure the eMMC has our partition layout (rewriting the GPT and resetting
# once if not). If Linux asked for DFU mode (edison-dfu sets dfu_once), clear the request
# and wait in DFU mode until flashed. Otherwise open a 5-second DFU window, then load
# vmlinuz from the FAT boot partition and boot it. If that fails, stay in DFU mode so the
# board can be reflashed.

bootdelay=1
baudrate=115200
loadaddr=0x100000

bootargs=root=PARTLABEL=rootfs rootfstype=ext4 rootwait console=ttyS2,115200n8 earlyprintk=ttyS2,115200n8,keep loglevel=4

bootcmd=run do_layout; run do_dfu_requested; run do_probe_dfu; run do_boot; echo "Boot failed, waiting in DFU mode (Ctrl+C for a prompt)"; run do_dfu

do_dfu_requested=if test "${dfu_once}" = 1; then setenv dfu_once; saveenv; echo "DFU mode requested from Linux: waiting for dfu-util (Ctrl+C for a prompt)"; run do_dfu; fi

do_boot=fatload mmc 0:7 ${loadaddr} vmlinuz && zboot ${loadaddr}

# Partition layout. 1-6 are fixed (U-Boot, its environment and the factory data must stay
# where the IFWI and tools expect them). config holds settings that survive reflashing;
# rootfs takes the rest of the eMMC. gpt verify compares names and sizes, so the GPT is only
# rewritten when the layout differs (first flash of this image); gpt write keeps the OSIP
# header in sector 0.
partitions=uuid_disk=${uuid_disk};name=u-boot0,start=1MiB,size=2MiB,uuid=${uuid_uboot0};name=u-boot-env0,size=1MiB,uuid=${uuid_uboot_env0};name=u-boot1,size=2MiB,uuid=${uuid_uboot1};name=u-boot-env1,size=1MiB,uuid=${uuid_uboot_env1};name=factory,size=1MiB,uuid=${uuid_factory};name=panic,size=24MiB,uuid=${uuid_panic};name=boot,size=64MiB,uuid=${uuid_boot};name=config,size=64MiB,uuid=${uuid_config};name=rootfs,size=-,uuid=${uuid_rootfs};
do_layout=if gpt verify mmc 0 ${partitions}; then echo "Partition layout OK"; else echo "Writing edison-revival partition layout"; gpt write mmc 0 ${partitions} && reset; fi

dfu_to_sec=5
do_dfu_alt_info_mmc=setenv dfu_alt_info "ifwi${hardware_id} raw 0 8192 mmcpart 1;ifwib${hardware_id} raw 0 8192 mmcpart 2;u-boot0 part 0 1;u-boot-env0 part 0 2;u-boot1 part 0 3;u-boot-env1 part 0 4;boot part 0 7;config part 0 8;rootfs part 0 9"
do_probe_dfu=run do_dfu_alt_info_mmc; dfu 0 mmc 0 ${dfu_to_sec}
do_dfu=run do_dfu_alt_info_mmc; dfu 0 mmc 0
